## P1 — 自适应学习

### LING-P1-ADAPT-1 · Spaced Repetition 算法

**目标**: 根据遗忘曲线自动安排复习，而非按固定顺序

**算法**: Leitner Box（简化版 SM-18）

```typescript
// lib/spaced-repetition.ts

type CardBox = 0 | 1 | 2 | 3 | 4; // 0=新词, 4=已掌握

interface WordCard {
  word: string;
  box: CardBox;
  nextReview: Date;
  easeFactor: number; // 2.5 默认
}

const BOX_INTERVALS = [1, 3, 7, 14, 30]; // 天

export function 计算下次复习时间(
  card: WordCard,
  correct: boolean
): WordCard {
  if (correct) {
    const newBox = Math.min(card.box + 1, 4) as CardBox;
    const interval = BOX_INTERVALS[newBox] * card.easeFactor;
    return {
      ...card,
      box: newBox,
      nextReview: addDays(new Date(), interval),
      easeFactor: Math.min(card.easeFactor + 0.1, 2.5),
    };
  } else {
    // 答错，降级到 Box 0
    return {
      ...card,
      box: 0,
      nextReview: new Date(),
      easeFactor: Math.max(card.easeFactor - 0.2, 1.3),
    };
  }
}
```
